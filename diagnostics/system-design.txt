calculate node resource usage 
    every time task is scheduled, add processing time to value called CPU time
    percentage is therefore = CPU_time_ms / sample_delay / (CPU_cores * 1000) * 100


cpu_time = 65_000 
sample_delay = 4
cpu_time_per_second = 65_000 // 4 = 16_000 ms per s

available processing time = cores * 1000
= 16 * 1000 = 16,000

cpu_time_per_second / available_processing_time = 16_000 / 16_000 = 1

(cpu_time // sample_delay) // (cores * 1000) * 100
